# ============================================================
# Milvus（可选）
# ============================================================
# 只有 Schema 索引（schema_indexing、HybridSchemaRetrievalService.from_milvus）

# 版本必须锁死：

#        MilvusException: (code=6, message=schema has no vector field ...)
#      schema_indexing/milvus_client.py 的 relation_collection 只有 VARCHAR 字段，
#      新版 Milvus 不再允许这种表结构
#   2) milvus-lite 2.5.1 还在 import pkg_resources，setuptools 81 起会移除它
pymilvus==2.5.18
milvus-lite==2.5.1
setuptools<81
